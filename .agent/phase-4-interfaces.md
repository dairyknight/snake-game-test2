## src/game/Snake.ts

```typescript
import { Vector2 } from '@/utils/Vector2';

export type Direction = Vector2;

export const UP:    Direction = new Vector2( 0, -1);
export const DOWN:  Direction = new Vector2( 0, +1);
export const LEFT:  Direction = new Vector2(-1,  0);
export const RIGHT: Direction = new Vector2(+1,  0);

export class Snake {
  /**
   * Constructs a snake with 3 segments at the board centre, moving RIGHT.
   * Initial segments: head=(10,9), (9,9), (8,9).
   */
  constructor();

  /**
   * The current head position (segments[0]).
   * @throws Error if segments array is empty (invariant violation).
   */
  get head(): Vector2;

  /**
   * Readonly snapshot of all segments, head first.
   */
  get segments(): readonly Vector2[];

  /**
   * The direction the snake moved on its last tick.
   * Initialised to RIGHT.
   */
  get currentDirection(): Direction;

  /**
   * Advance the snake one step in `direction`.
   *
   * Rules:
   * - If `direction` is the 180° opposite of `currentDirection`, it is silently
   *   ignored and the snake continues in `currentDirection`.
   * - A new head is prepended at head.add(effectiveDirection).
   * - If `_growPending` is true, the tail is NOT removed and the flag is cleared.
   * - Otherwise the tail is removed (length stays constant).
   * - `currentDirection` is updated to `effectiveDirection`.
   */
  move(direction: Direction): void;

  /**
   * Schedule one segment of growth on the next move().
   * Sets an internal boolean flag; multiple consecutive calls before the next
   * move() still only grow by 1 (flag, not counter).
   */
  grow(): void;
}
```

### Invariant diagram

```
Before move(RIGHT):
  segments = [(10,9), (9,9), (8,9)]   currentDirection = RIGHT
  _growPending = false

After move(RIGHT):
  segments = [(11,9), (10,9), (9,9)]  currentDirection = RIGHT
  tail (8,9) removed

Before move(RIGHT) with _growPending=true:
  segments = [(10,9), (9,9), (8,9)]
  _growPending = true

After move(RIGHT) with _growPending=true:
  segments = [(11,9), (10,9), (9,9), (8,9)]   length = 4
  _growPending = false

180° reversal: move(LEFT) when currentDirection=RIGHT:
  effectiveDirection = RIGHT (LEFT ignored)
  segments = [(11,9), (10,9), (9,9)]
```

## src/game/Board.ts

```typescript
import { Vector2 } from '@/utils/Vector2';
import { Snake } from '@/game/Snake';

export class Board {
  readonly width: number;
  readonly height: number;

  /**
   * @param width  Number of columns (default 20)
   * @param height Number of rows    (default 20)
   */
  constructor(width?: number, height?: number);

  /**
   * Returns true if `pos` represents a collision (wall OR self).
   *
   * Wall collision: pos.x < 0 || pos.x >= width || pos.y < 0 || pos.y >= height
   *
   * Self-collision: pos equals any element of snake.segments EXCEPT segments[0]
   *   (segments[0] is the current head — the snake cannot collide with its own
   *   leading edge; it can only collide with its body/tail).
   *
   * Called AFTER snake.move() — snake.segments already includes the new head.
   * The skip-head rule prevents segments[0] from trivially matching `pos`.
   *
   * @param pos   The new head position to check (= snake.head after move)
   * @param snake The snake to check self-collision against
   */
  checkCollision(pos: Vector2, snake: Snake): boolean;
}
```

### Collision decision table

```
pos          | in bounds? | on body[1..]? | result
─────────────┼────────────┼───────────────┼────────
(-1, 10)     | NO         | —             | true  (wall)
(20, 10)     | NO         | —             | true  (wall)
(10, -1)     | NO         | —             | true  (wall)
(10, 20)     | NO         | —             | true  (wall)
(0, 0)       | YES        | NO            | false
(19, 19)     | YES        | NO            | false
(9, 9)       | YES        | YES (body)    | true  (self)
(10, 9)      | YES        | NO (is head)  | false (skip)
```

## src/game/Food.ts

```typescript
import { Vector2 } from '@/utils/Vector2';

export class Food {
  /**
   * Constructs a Food item and immediately spawns at a random unoccupied cell.
   *
   * @param boardWidth   Board width (default 20)
   * @param boardHeight  Board height (default 20)
   * @param occupied     Initial occupied cells (default [])
   * @param rng          Random number generator, returns [0,1) (default Math.random).
   *                     Injected for deterministic testing.
   */
  constructor(
    boardWidth?: number,
    boardHeight?: number,
    occupied?: Vector2[],
    rng?: () => number,
  );

  /**
   * Current food position. Always a valid unoccupied cell (set at construction
   * and after each successful respawn).
   */
  get position(): Vector2;

  /**
   * Choose a new random position not in `occupied`.
   *
   * Algorithm:
   *   1. Build available[] = all board cells not in occupied (uses equals() comparison)
   *   2. If available.length === 0: return false (board full — win condition stub)
   *   3. Pick available[Math.floor(rng() * available.length)]
   *   4. Update internal position, return true
   *
   * @param occupied  Array of cells to avoid (typically snake.segments)
   * @returns true if a new position was set; false if no cells available (board full)
   */
  respawn(occupied: Vector2[]): boolean;
}
```

### update() body

```typescript
update(_deltaTime?: number): void {
  if (this._state !== GameState.PLAYING) return;

  // Apply queued direction (or continue current direction)
  const dir = this._pendingDirection ?? this._snake.currentDirection;
  this._pendingDirection = null;

  this._snake.move(dir);

  // Collision: wall or self (checked after move, skips head for self-check)
  if (this._board.checkCollision(this._snake.head, this._snake)) {
    this.endGame();
    return;
  }

  // Food eaten
  if (this._food.position.equals(this._snake.head)) {
    this._snake.grow();
    this._food.respawn(Array.from(this._snake.segments));
    this.events.emit('foodEaten', undefined);
  }
}
```

### restartGame() — TODO populated

restartGame(): void {
  if (this._state !== GameState.GAME_OVER) return;
  // Reset entities to initial state
  this._snake = new Snake();
  this._food = new Food(this._board.width, this._board.height, Array.from(this._snake.segments));

}

## Wiring diagram (GameManager.update)

```
GameManager.update()
  │
  ├─ guard: state !== PLAYING → return (no-op)
  │
  ├─ effectiveDir = _pendingDirection ?? _snake.currentDirection
  │   then: _pendingDirection = null
  │
  ├─ _snake.move(effectiveDir)
  │   ├─ 180° check → use currentDirection if opposite
  │   ├─ prepend new head
  │   └─ pop tail (unless _growPending → retain + clear flag)
  │
  ├─ _board.checkCollision(_snake.head, _snake)
  │   ├─ wall:  pos outside [0,width) × [0,height) → true
  │   ├─ self:  pos equals any segments[1..] → true
  │   └─ false → continue
  │       └─ true → endGame() → GAME_OVER → return
  │
  └─ _food.position.equals(_snake.head)?
      ├─ yes → _snake.grow()
      │        _food.respawn(snake.segments)  [returns bool; false = board full, no-op]
      │        events.emit('foodEaten', undefined)
      └─ no  → (nothing)
```
